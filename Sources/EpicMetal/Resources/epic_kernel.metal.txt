#include <metal_stdlib>
using namespace metal;

// ===========================================================================
// Epic Cash ProgPoW 0.9.2 search kernel.
//
// Derived from kawpow-mac/Sources/kawpow-mac/Resources/kernel_template.metal
// (GPLv3). The Metal scaffolding that is variant-independent is preserved:
//   - threadgroup `c_dag` preload (16 KiB)
//   - the h = 0..15 nonce-transposition loop (16 hashes per 16-lane group)
//   - cross-lane FNV reduction into the 8-word mix digest
//   - the atomic "first writer wins" result claim
//
// What is Epic-specific and had to change (see EPIC_METAL_PROGPOW_AUDIT.md §5):
//   1. keccak-f800 state layouts for BOTH permutations (standard constants,
//      zero padding instead of ravencoin_rndc)
//   2. runtime seed derivation: seed_lo = bswap32(st[1]), seed_hi = bswap32(st[0])
//   3. second permutation uses the ORIGINAL header words, not the permuted state
//   4. full 64-bit nonce (KawPow hashes only the low 32 bits)
//   5. target / dag_elements / base nonce are runtime arguments, not baked
//      into the source, because Epic's difficulty changes every block
//
// Authoritative reference: EpicCash/progpow-rust pp_light/src/progpow.rs
// ===========================================================================

#define GROUP_SIZE 256
#define PROGPOW_LANES 16
#define PROGPOW_REGS 32
#define PROGPOW_DAG_LOADS 4
#define PROGPOW_CACHE_WORDS 4096
#define PROGPOW_CNT_DAG 64
#define PROGPOW_CNT_MATH 20
#define HASHES_PER_GROUP (GROUP_SIZE / PROGPOW_LANES)
#define FNV_PRIME 0x1000193
#define FNV_OFFSET_BASIS 0x811c9dc5

typedef struct { unsigned int s[PROGPOW_DAG_LOADS]; } dag_t;

// Standard keccak-f[800] round constants (progpow.rs:50-54). NOT Ravencoin's
// "rAVENCOINKAWPOW" ASCII table.
constant unsigned int keccakf_rndc[24] = {
    0x00000001, 0x00008082, 0x0000808a, 0x80008000,
    0x0000808b, 0x80000001, 0x80008081, 0x00008009,
    0x0000008a, 0x00000088, 0x80008009, 0x8000000a,
    0x8000808b, 0x0000008b, 0x00008089, 0x00008003,
    0x00008002, 0x00000080, 0x0000800a, 0x8000000a,
    0x80008081, 0x00008080, 0x80000001, 0x80008008
};

#define ROTL32(x, n) rotate((x), (unsigned int)(n))
#define ROTR32(x, n) rotate((x), (unsigned int)(32-n))

// ---------------------------------------------------------------------------
// keccak-f[800]: 25 x 32-bit lanes, 22 rounds. Identical to the reference
// keccakf800.c used by ProgPoW (progpow.rs:64-102).
// ---------------------------------------------------------------------------
void keccak_f800_round(thread unsigned int st[25], const int r)
{
    const unsigned int keccakf_rotc[24] = {
        1,3,6,10,15,21,28,36,45,55,2,14,27,41,56,8,25,43,62,18,39,61,20,44};
    const unsigned int keccakf_piln[24] = {
        10,7,11,17,18,3,5,16,8,21,24,4,15,23,19,13,12,2,20,14,22,9,6,1};
    unsigned int t, bc[5];
    for (int i = 0; i < 5; i++)
        bc[i] = st[i] ^ st[i+5] ^ st[i+10] ^ st[i+15] ^ st[i+20];
    for (int i = 0; i < 5; i++) {
        t = bc[(i+4) % 5] ^ ROTL32(bc[(i+1) % 5], 1u);
        for (unsigned int j = 0; j < 25; j += 5)
            st[j+i] ^= t;
    }
    t = st[1];
    for (int i = 0; i < 24; i++) {
        unsigned int j = keccakf_piln[i];
        bc[0] = st[j];
        st[j] = ROTL32(t, keccakf_rotc[i]);
        t = bc[0];
    }
    for (unsigned int j = 0; j < 25; j += 5) {
        for (int i = 0; i < 5; i++) bc[i] = st[j+i];
        for (int i = 0; i < 5; i++) st[j+i] ^= (~bc[(i+1) % 5]) & bc[(i+2) % 5];
    }
    st[0] ^= keccakf_rndc[r];
}

void keccak_f800(thread unsigned int* st)
{
    for (int r = 0; r < 22; r++) {
        keccak_f800_round(st, r);
    }
}

#define fnv1a(h, d) (h = (h ^ d) * FNV_PRIME)

// Byte-reverse a 32-bit word. Epic derives its per-hash seed as
// seed = (bswap32(st[0]) << 32) | bswap32(st[1])  (progpow.rs:127).
inline unsigned int bswap32(unsigned int x)
{
    return ((x & 0x000000ffu) << 24) |
           ((x & 0x0000ff00u) <<  8) |
           ((x & 0x00ff0000u) >>  8) |
           ((x & 0xff000000u) >> 24);
}

inline unsigned long bswap64(unsigned long x)
{
    x = (x & 0x00000000FFFFFFFFul) << 32 | (x & 0xFFFFFFFF00000000ul) >> 32;
    x = (x & 0x0000FFFF0000FFFFul) << 16 | (x & 0xFFFF0000FFFF0000ul) >> 16;
    x = (x & 0x00FF00FF00FF00FFul) <<  8 | (x & 0xFF00FF00FF00FF00ul) >>  8;
    return x;
}

// ---------------------------------------------------------------------------
// kiss99 PRNG + per-lane mix initialisation (progpow.rs:143-192).
// ---------------------------------------------------------------------------
typedef struct { unsigned int z, w, jsr, jcong; } kiss99_t;

unsigned int kiss99(thread kiss99_t* st)
{
    st->z = 36969 * (st->z & 65535) + (st->z >> 16);
    st->w = 18000 * (st->w & 65535) + (st->w >> 16);
    unsigned int MWC = ((st->z << 16) + st->w);
    st->jsr ^= (st->jsr << 17);
    st->jsr ^= (st->jsr >> 13);
    st->jsr ^= (st->jsr << 5);
    st->jcong = 69069 * st->jcong + 1234567;
    return ((MWC ^ st->jcong) + st->jsr);
}

// seed[0] = low 32 bits of the ProgPoW seed, seed[1] = high 32 bits.
// fill_mix chains fnv1a from FNV_OFFSET_BASIS through seed_lo, seed_hi,
// lane_id, lane_id — identical to progpow-rust fill_mix (progpow.rs:174-192).
void fill_mix(threadgroup unsigned int* seed, unsigned int lane_id, thread unsigned int* a_mix)
{
    unsigned int fnv_hash = FNV_OFFSET_BASIS;
    kiss99_t st;
    st.z     = fnv1a(fnv_hash, seed[0]);
    st.w     = fnv1a(fnv_hash, seed[1]);
    st.jsr   = fnv1a(fnv_hash, lane_id);
    st.jcong = fnv1a(fnv_hash, lane_id);
    for (int i = 0; i < PROGPOW_REGS; i++)
        a_mix[i] = kiss99(&st);
}

typedef struct { unsigned int uint32s[PROGPOW_LANES]; } shuffle_t;
typedef struct { unsigned int uint32s[8]; } hash32_t;

// ---------------------------------------------------------------------------
// epic_search — ProgPoW 0.9.2 search.
//
// Buffers:
//   0 g_dag          the epoch DAG, as 16-byte dag_t elements
//   1 header_words   8 words = keccak_256(pre_pow) read as little-endian u32
//   2 results        output; see layout below
//   3 resultsIndex   atomic claim counter (host zeroes per dispatch)
//   4 stop           0/1; set to 1 by the first winner, also polled on entry
//   5 baseNonce      full 64-bit nonce of grid position 0
//   6 target         top-64-bit prefilter threshold (mining mode only)
//   7 dagElements    data_size / 256 (ProgPoW's PROGPOW_DAG_ELEMENTS)
//   8 probeMode      0 = mining, non-zero = known-answer probe
//   9 probeNonce     the nonce to report in probe mode
//
// results layout
//   mining: [0]=nonce lo  [1]=nonce hi  [2..9]=mix words  [10]=result hi
//           [11]=result lo
//   probe:  [0]=nonce lo  [1]=nonce hi  [2..9]=mix words  [10..17]=final words
// ---------------------------------------------------------------------------
kernel void epic_search(
                   device const dag_t * g_dag,
                   constant uint *header_words,
                   device uint *results,
                   device atomic_uint *resultsIndex,
                   device uint *stop,
                   constant ulong &baseNonce,
                   constant ulong &target,
                   constant uint &dagElements,
                   constant uint &probeMode,
                   constant ulong &probeNonce,
                   uint nonceOffset [[thread_position_in_grid]],
                   uint lid [[thread_position_in_threadgroup]]
                )
{
    const unsigned int lane_id  = lid & (PROGPOW_LANES - 1);
    const unsigned int group_id = lid / PROGPOW_LANES;

    if (stop[0]) {
        return;
    }

    threadgroup shuffle_t share[HASHES_PER_GROUP];
    threadgroup unsigned int c_dag[PROGPOW_CACHE_WORDS];

    // Preload the first 16 KiB of the DAG (= DAG items 0..255) into threadgroup
    // memory. This is exactly ProgPoW's c_dag; it needs no separate build.
    for (unsigned int word = lid * PROGPOW_DAG_LOADS; word < PROGPOW_CACHE_WORDS;
         word += GROUP_SIZE * PROGPOW_DAG_LOADS)
    {
        dag_t load = g_dag[word / PROGPOW_DAG_LOADS];
        for (int i = 0; i < PROGPOW_DAG_LOADS; i++)
            c_dag[word + i] = load.s[i];
    }

    // Full 64-bit nonce. baseNonce is the nonce of grid position 0, so the
    // 2^32 carry is handled by 64-bit arithmetic (KawPow truncated this).
    const unsigned long nonce = baseNonce + (unsigned long)nonceOffset;
    const unsigned int nonce_lo = (unsigned int)nonce;
    const unsigned int nonce_hi = (unsigned int)(nonce >> 32);

    // ---- first keccak-f800 permutation (progpow.rs:104-122) ---------------
    // st[0..7] = header hash, st[8..9] = nonce, st[10..24] = 0
    unsigned int state_a[25];
    for (int i = 0; i < 8; i++)
        state_a[i] = header_words[i];
    state_a[8] = nonce_lo;
    state_a[9] = nonce_hi;
    for (int i = 10; i < 25; i++)
        state_a[i] = 0;
    keccak_f800(state_a);

    // Runtime seed for fill_mix (progpow.rs:124-128):
    //   seed = (bswap32(st[0]) << 32) | bswap32(st[1])
    // i.e. seed_lo = bswap32(st[1]), seed_hi = bswap32(st[0]).
    // KawPow instead uses state2[0]/state2[1] unswapped and in the other order.
    const unsigned int seed_lo = bswap32(state_a[1]);
    const unsigned int seed_hi = bswap32(state_a[0]);

    // ---- h loop: nonce transposition ------------------------------------
    // In iteration h the 16-lane group hashes the nonce owned by lane h, and
    // lane h captures the digest. Over h = 0..15 every lane ends up holding the
    // digest of its own nonce.
    hash32_t digest;
    for (unsigned int h = 0; h < PROGPOW_LANES; h++)
    {
        unsigned int the_mix[PROGPOW_REGS];
        if (lane_id == h) {
            share[group_id].uint32s[0] = seed_lo;
            share[group_id].uint32s[1] = seed_hi;
        }
        threadgroup_barrier(mem_flags::mem_threadgroup);
        fill_mix(share[group_id].uint32s, lane_id, the_mix);
        for (unsigned int loop = 0; loop < PROGPOW_CNT_DAG; ++loop)
        {
            if (lane_id == (loop % PROGPOW_LANES))
                share[0].uint32s[group_id] = the_mix[0];
            threadgroup_barrier(mem_flags::mem_threadgroup);
            unsigned int offset = share[0].uint32s[group_id];
            offset %= dagElements;
            offset = offset * PROGPOW_LANES + (lane_id ^ loop) % PROGPOW_LANES;
            dag_t data_dag = g_dag[offset];
#if !defined(EPIC_EXPERIMENTAL_NO_POST_DAG_BARRIER)
            threadgroup_barrier(mem_flags::mem_threadgroup);
#endif
            unsigned int data;
            INCLUDE_PROGPOW_RANDOM_MATH
            threadgroup_barrier(mem_flags::mem_threadgroup);
            INCLUDE_PROGPOW_DATA_LOADS
        }
        unsigned int mix_hash = FNV_OFFSET_BASIS;
        for (int i = 0; i < PROGPOW_REGS; i++)
            fnv1a(mix_hash, the_mix[i]);
        hash32_t digest_temp;
        for (int i = 0; i < 8; i++)
            digest_temp.uint32s[i] = FNV_OFFSET_BASIS;
        share[group_id].uint32s[lane_id] = mix_hash;
        threadgroup_barrier(mem_flags::mem_threadgroup);
        for (int i = 0; i < PROGPOW_LANES; i++)
            fnv1a(digest_temp.uint32s[i % 8], share[group_id].uint32s[i]);
        if (h == lane_id)
            digest = digest_temp;
        // Barrier at the END of the iteration: without it, the next
        // iteration's write to share[group_id].uint32s[0..1] can race with a
        // slower lane still reading those words in the reduction above.
        threadgroup_barrier(mem_flags::mem_threadgroup);
    }

    // ---- second keccak-f800 permutation (progpow.rs:130-136, 387) --------
    // st[0..7] = ORIGINAL header hash (not the permuted state), st[8..9] = seed,
    // st[10..17] = mix digest, st[18..24] = 0.
    unsigned int state_b[25];
    for (int i = 0; i < 8; i++)
        state_b[i] = header_words[i];
    state_b[8] = seed_lo;
    state_b[9] = seed_hi;
    for (int i = 0; i < 8; i++)
        state_b[10 + i] = digest.uint32s[i];
    for (int i = 18; i < 25; i++)
        state_b[i] = 0;
    keccak_f800(state_b);

    // Top 64 bits of the big-endian final hash (progpow.rs / epic-miner
    // miner.rs:102-106): bswap64(st[1] << 32 | st[0]).
    const unsigned long result = bswap64(((unsigned long)state_b[1] << 32) | state_b[0]);

    if (probeMode != 0u) {
        // Known-answer probe: report the full mix + final for one nonce.
        if (nonce == probeNonce) {
            uint claimed = atomic_fetch_add_explicit(resultsIndex, 1u, memory_order_relaxed);
            if (claimed == 0u) {
                results[0] = nonce_lo;
                results[1] = nonce_hi;
                for (int i = 0; i < 8; i++) results[2 + i]  = digest.uint32s[i];
                for (int i = 0; i < 8; i++) results[10 + i] = state_b[i];
                stop[0] = 1u;
            }
        }
    } else if (result <= target) {
        // First winning thread in this dispatch writes; without the claim two
        // threads can interleave a nonce with the wrong mix.
        uint claimed = atomic_fetch_add_explicit(resultsIndex, 1u, memory_order_relaxed);
        if (claimed == 0u) {
            results[0] = nonce_lo;
            results[1] = nonce_hi;
            for (int i = 0; i < 8; i++) results[2 + i] = digest.uint32s[i];
            // Return every final-digest word, not just the 64-bit prefilter.
            // The host must compare this GPU-produced full 256-bit value to
            // Epic's exact share boundary before it serializes a submission.
            for (int i = 0; i < 8; i++) results[10 + i] = state_b[i];
            stop[0] = 1u;
        }
    }
}
